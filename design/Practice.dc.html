# practices: xkeeper-pg/device-meta@v3
import os, psycopg

def run(imei: str):
    """Метаданные прибора по IMEI: модель, дата создания, сервер."""
    dsn = (f"host={os.environ['XK_PG_HOST']} port={os.environ['XK_PG_PORT_DEVICES']} "
           f"dbname=x-keeper_devices user={os.environ['XK_PG_USER']} "
           f"password={os.environ['XK_PG_PASSWORD']} sslmode=verify-ca")
    with psycopg.connect(dsn, options="-c default_transaction_read_only=on") as conn:
        row = conn.execute("""
            SELECT d.imei, d.model_id, d.create_dt, d.server_id
            FROM devices d
            WHERE d.imei = %s
        """, (imei,)).fetchone()
    return None if row is None else dict(zip(("imei", "model_id", "create_dt", "server_id"), row))
